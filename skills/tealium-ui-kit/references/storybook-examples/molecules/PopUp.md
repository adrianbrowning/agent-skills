# PopUp

## Example Usage

```tsx
import { PopUp } from '@tealiumlabs/tealium-ui-kit/molecules/PopUp';

function Example() {
  return <PopUp buttonType="primary"
      buttonText="Click Me"
      popUpText="PopUpText" />;
}
```
