# Button

## Example Usage

```tsx
import { Button } from '@tealiumlabs/tealium-ui-kit/atoms/button';

function Example() {
  return <Button children="Button"
      type="primary" />;
}
```

### Secondary Example

```tsx
import { Button } from '@tealiumlabs/tealium-ui-kit/atoms/button';

function SecondaryExample() {
  return (
    <Button
      children="Button"
      type="secondary"
    />
  );
}
```
