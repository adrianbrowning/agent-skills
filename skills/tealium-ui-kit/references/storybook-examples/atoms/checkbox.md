# Checkbox

## Example Usage

```tsx
import { Checkbox } from '@tealiumlabs/tealium-ui-kit/atoms/checkbox';

function Example() {
  return (
    <Checkbox
      label="Checkbox"
      id="checkboxId"
      inputProps={...}
      "aria-label"="Checkbox"
    />
  );
}
```
